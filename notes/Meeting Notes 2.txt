def guess_dataset_name(h5_path: str) -> str:
    """Pick the first top-level dataset/group name in the OpenMC mgxs.h5."""
    with h5py.File(h5_path, "r") as f:
        keys = list(f.keys())
    if not keys:
        raise RuntimeError(f"No top-level datasets found in {h5_path}")
    return keys[0]
